cmake_minimum_required(VERSION 3.20)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" IB_VERSION LIMIT_COUNT 1)
if(NOT IB_VERSION MATCHES "^[0-9]+\\.[0-9]+\\.[0-9]+$")
    message(FATAL_ERROR "VERSION must contain a version in MAJOR.MINOR.PATCH format")
endif()
project(installbuilder_unpacker VERSION ${IB_VERSION} LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    add_compile_options(/utf-8)
    add_compile_definitions(_CRT_SECURE_NO_WARNINGS)
endif()
if(WIN32)
    add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
endif()
find_package(ZLIB REQUIRED)
find_package(LibLZMA REQUIRED)
find_package(Threads REQUIRED)
include(CheckTypeSize)
include(CheckIncludeFileCXX)
check_type_size("long" SIZEOF_LONG LANGUAGE CXX)
check_include_file_cxx("unistd.h" HAVE_UNISTD_H)
if(UNIX)
    set(HAVE_MMAP 1)
endif()
configure_file(src/metakit_config.h.in generated/config.h)
configure_file(src/version.h.in generated/version.h @ONLY)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS VERSION)

set(MK_SOURCES column custom derived fileio field format handler persist remap std store string table univ view viewx)
list(TRANSFORM MK_SOURCES PREPEND "third_party/metakit/src/")
list(TRANSFORM MK_SOURCES APPEND ".cpp")
add_library(metakit STATIC ${MK_SOURCES})
target_include_directories(metakit SYSTEM PUBLIC third_party/metakit/include)
target_include_directories(metakit PRIVATE third_party/metakit/src "${CMAKE_CURRENT_BINARY_DIR}/generated")
if(SIZEOF_LONG EQUAL 8)
    target_compile_definitions(metakit PUBLIC q4_LONG64=1)
endif()
set_target_properties(metakit PROPERTIES CXX_STANDARD 11)

file(GLOB LZHAM_SOURCES CONFIGURE_DEPENDS third_party/lzham/lzhamdecomp/*.cpp)
add_library(lzham_decoder STATIC ${LZHAM_SOURCES})
target_include_directories(lzham_decoder PRIVATE third_party/lzham/include third_party/lzham/lzhamdecomp)
if(NOT MSVC)
    target_compile_options(lzham_decoder PRIVATE -fno-strict-aliasing)
endif()
target_compile_definitions(lzham_decoder PRIVATE LZHAM_ANSI_CPLUSPLUS)

add_library(twofish STATIC third_party/libtomcrypt/twofish.c)
target_include_directories(twofish PUBLIC third_party/libtomcrypt/headers)
target_compile_definitions(twofish PUBLIC LTC_NOTHING LTC_TWOFISH LTC_NO_TEST LTC_NO_ASM ARGTYPE=4)

add_library(ib_core STATIC src/archive.cpp src/tcl_list.cpp src/codecs.cpp src/cookfs.cpp src/extract.cpp src/platform.cpp)
target_include_directories(ib_core PUBLIC src PRIVATE third_party/picosha2)
target_link_libraries(ib_core PUBLIC metakit ZLIB::ZLIB LibLZMA::LibLZMA Threads::Threads twofish lzham_decoder)
if(WIN32 AND VCPKG_TARGET_TRIPLET MATCHES "-static")
    target_compile_definitions(ib_core PRIVATE LZMA_API_STATIC)
endif()
if(MSVC)
    target_compile_options(ib_core PRIVATE /W4)
else()
    target_compile_options(ib_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(ib-unpack src/main.cpp src/progress.cpp)
target_include_directories(ib-unpack PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(ib-unpack PRIVATE ib_core)
if(WIN32)
    enable_language(RC)
    configure_file(src/windows.rc.in generated/windows.rc @ONLY)
    target_sources(ib-unpack PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated/windows.rc")
    target_link_libraries(ib-unpack PRIVATE shell32)
    if(MSVC)
        target_link_options(ib-unpack PRIVATE /MANIFEST:NO)
    endif()
endif()
install(TARGETS ib-unpack RUNTIME DESTINATION bin)
install(FILES README.md VERSION LICENSE DESTINATION share/ib-unpack)
install(FILES third_party/metakit/license.terms DESTINATION share/ib-unpack/licenses/metakit)
install(FILES third_party/lzham/LICENSE DESTINATION share/ib-unpack/licenses/lzham)
install(FILES third_party/libtomcrypt/LICENSE DESTINATION share/ib-unpack/licenses/libtomcrypt)
install(FILES third_party/picosha2/LICENSE DESTINATION share/ib-unpack/licenses/picosha2)

if(WIN32 AND DEFINED VCPKG_INSTALLED_DIR)
    foreach(dependency zlib liblzma)
        install(FILES "${VCPKG_INSTALLED_DIR}/${VCPKG_TARGET_TRIPLET}/share/${dependency}/copyright"
                DESTINATION "share/ib-unpack/licenses/${dependency}")
    endforeach()
endif()

set(CPACK_PACKAGE_NAME "ib-unpack")
set(CPACK_PACKAGE_VERSION "${PROJECT_VERSION}")
set(IB_PACKAGE_ARCH "${CMAKE_SYSTEM_PROCESSOR}" CACHE STRING "Architecture label for release packages")
string(TOLOWER "${CMAKE_SYSTEM_NAME}" IB_PACKAGE_OS)
if(APPLE)
    set(IB_PACKAGE_OS "macos")
endif()
set(CPACK_PACKAGE_FILE_NAME "ib-unpack-${PROJECT_VERSION}-${IB_PACKAGE_OS}-${IB_PACKAGE_ARCH}")
if(WIN32)
    set(CPACK_GENERATOR ZIP)
else()
    set(CPACK_GENERATOR TGZ)
endif()
include(CPack)
